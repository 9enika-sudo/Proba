<script>
let entries = JSON.parse(localStorage.getItem('bodyEntries') || '[]');
let height = parseFloat(localStorage.getItem('bodyHeight')) || null;
document.getElementById('date').valueAsDate = new Date();
if (height) document.getElementById('height').value = height;
function saveHeight() {
const h = parseFloat(document.getElementById('height').value);
if (!h || h < 100 || h > 250) {
alert('Вкажи коректний зріст (100–250 см)');
return;
}
height = h;
localStorage.setItem('bodyHeight', h);
render();
alert('Зріст збережено!');
}
function addEntry() {
const date = document.getElementById('date').value;
const weight = parseFloat(document.getElementById('weight').value);
if (!date || !weight) {
alert('Дата і вага обов’язкові');
return;
}
const entry = {
id: Date.now(),
date,
weight,
waist: parseFloat(document.getElementById('waist').value) || null,
hips: parseFloat(document.getElementById('hips').value) || null,
chest: parseFloat(document.getElementById('chest').value) || null,
biceps: parseFloat(document.getElementById('biceps').value) || null,
thigh: parseFloat(document.getElementById('thigh').value) || null,
bodyfat: parseFloat(document.getElementById('bodyfat').value) || null,
notes: document.getElementById('notes').value.trim()
};
entries.push(entry);
entries.sort((a, b) => a.date.localeCompare(b.date));
localStorage.setItem('bodyEntries', JSON.stringify(entries));
['weight','waist','hips','chest','biceps','thigh','bodyfat','notes'].forEach(id => {
document.getElementById(id).value = '';
});
document.getElementById('date').valueAsDate = new Date();
render();
}
function deleteEntry(id) {
if (!confirm('Видалити цей вимір?')) return;
entries = entries.filter(e => e.id !== id);
localStorage.setItem('bodyEntries', JSON.stringify(entries));
render();
}
function calcBMI(weight) {
if (!height || !weight) return null;
const h = height / 100;
return (weight / (h * h)).toFixed(1);
}
function bmiCategory(bmi) {
if (bmi < 18.5) return 'Недостатня вага';
if (bmi < 25) return 'Норма';
if (bmi < 30) return 'Надмірна вага';
return 'Ожиріння';
}
let weightChart, measuresChart;
function render() {
renderStats();
renderHistory();
renderCharts();
}
function renderStats() {
const el = document.getElementById('stats');
if (entries.length === 0) {
el.innerHTML = '';
return;
}
const last = entries[entries.length - 1];
const first = entries[0];
const weightDiff = (last.weight - first.weight).toFixed(1);
const bmi = calcBMI(last.weight);
el.innerHTML = `
<div class="stat">
<div class="stat-value">${last.weight}</div>
<div class="stat-label">Остання вага</div>
</div>
<div class="stat">
<div class="stat-value" style="color:${weightDiff > 0 ? 'var(--danger)' : 'var(--success)'}">
${weightDiff > 0 ? '+' : ''}${weightDiff}
</div>
<div class="stat-label">Зміна ваги</div>
</div>
${bmi ? `
<div class="stat">
<div class="stat-value">${bmi}</div>
<div class="stat-label">ІМТ · ${bmiCategory(bmi)}</div>
</div>` : ''}
<div class="stat">
<div class="stat-value">${entries.length}</div>
<div class="stat-label">Вимірів</div>
</div>
`;
}
function renderHistory() {
const el = document.getElementById('history');
if (entries.length === 0) {
el.innerHTML = '<div class="empty">Поки немає вимірів. Додай перший!</div>';
return;
}
let html = `<table>
<thead>
<tr>
<th>Дата</th>
<th>Вага</th>
<th>Талія</th>
<th>Стегна</th>
<th></th>
</tr>
</thead>
<tbody>`;
[...entries].reverse().forEach(e => {
html += `
<tr>
<td>${formatDate(e.date)}</td>
<td><strong>${e.weight}</strong></td>
<td>${e.waist || '—'}</td>
<td>${e.hips || '—'}</td>
<td><button class="btn btn-danger" onclick="deleteEntry(${e.id})">×</button></td>
</tr>`;
});
html += '</tbody></table>';
el.innerHTML = html;
}
function renderCharts() {
const labels = entries.map(e => formatDate(e.date));
const weightData = entries.map(e => e.weight);
if (weightChart) weightChart.destroy();
weightChart = new Chart(document.getElementById('weightChart'), {
type: 'line',
data: {
labels,
datasets: [{
label: 'Вага (кг)',
data: weightData,
borderColor: '#38bdf8',
backgroundColor: 'rgba(56, 189, 248, 0.1)',
fill: true,
tension: 0.3,
pointRadius: 4
}]
},
options: {
responsive: true,
maintainAspectRatio: false,
plugins: { legend: { display: false } },
scales: {
y: { grid: { color: '#334155' }, ticks: { color: '#94a3b8' } },
x: { grid: { display: false }, ticks: { color: '#94a3b8', maxRotation: 45 } }
}
}
});
if (measuresChart) measuresChart.destroy();
measuresChart = new Chart(document.getElementById('measuresChart'), {
type: 'line',
data: {
labels,
datasets: [
{
label: 'Талія',
data: entries.map(e => e.waist),
borderColor: '#f472b6',
tension: 0.3,
pointRadius: 3,
spanGaps: true
},
{
label: 'Стегна',
data: entries.map(e => e.hips),
borderColor: '#a78bfa',
tension: 0.3,
pointRadius: 3,
spanGaps: true
},
{
label: 'Груди',
data: entries.map(e => e.chest),
borderColor: '#34d399',
tension: 0.3,
pointRadius: 3,
spanGaps: true
}
]
},
options: {
responsive: true,
maintainAspectRatio: false,
plugins: {
legend: {
labels: { color: '#94a3b8', boxWidth: 12 }
}
},
scales: {
y: { grid: { color: '#334155' }, ticks: { color: '#94a3b8' } },
x: { grid: { display: false }, ticks: { color: '#94a3b8', maxRotation: 45 } }
}
}
});
}
function formatDate(dateStr) {
const d = new Date(dateStr);
return d.toLocaleDateString('uk-UA', { day: 'numeric', month: 'short' });
}
